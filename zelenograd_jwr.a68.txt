      INT p4 = (p3 > 0 | find(cmd, sil, p3) | 0);
      IF p1 = 0 OR p2 = 0 OR p3 = 0 THEN
        erreur("SyntaxError", "Chevrons Ω mal formés ou trait de fracture absent.",
               "La rune s'efface immédiatement. Goût métallique en bouche.",
               "Recommencer après 7 respirations.")
      ELIF p4 = 0 THEN
        erreur("SyntaxError", "Silence sacré (•••) absent : la compilation avorte.",
               "La rune s'efface immédiatement. Goût métallique en bouche.",
               "Recommencer après 7 respirations.")
      ELSE
        INT psp = find(cmd, " ", p2 + UPB sep);
        STRING glyphe = cmd[p1 + UPB omo : p2 - 1];
        INT version = toint(cmd[p2 + UPB sep : psp - 1]);
        STRING lieu = cmd[psp + 1 : p3 - 1];
        STRING intention = cmd[p3 + UPB fra : p4 - 1];
        INT ri = rune_index(glyphe), li = lieu_index(lieu);
        IF ri = 0 THEN
          erreur("SyntaxError", "Rune inconnue : « " + glyphe + " ».",
                 "La rune s'efface immédiatement. Goût métallique en bouche.",
                 "Recommencer après 7 respirations.")
        ELIF (ri = 8 AND (find(intention, "brûl", 1) > 0 OR find(intention, "feu", 1) > 0))
             OR (ri = 9 AND (find(intention, "figer", 1) > 0 OR find(intention, "geler", 1) > 0))
             OR (ri = 4 AND find(intention, "oubli", 1) > 0) THEN
          erreur("TypeMismatch", rn[ri] + " est incompatible avec « " + intention + " ».",
                 "Rejet violent. Migraine aiguë, saignement de nez.",
                 "Appliquer un galet froid sur le front.")
        ELIF li = 0 THEN
          erreur("LocationMismatch", "Lieu « " + lieu + " » inconnu : aucun ancrage tellurique.",
                 "La rune flotte dans l'air sans effet, puis se dissipe.",
                 "Se déplacer vers un lieu de résonance.")
        ELIF find(la[li], glyphe, 1) = 0 THEN
          erreur("LocationMismatch", rn[ri] + " ne résonne pas à " + lieu + ".",
                 "La rune flotte dans l'air sans effet, puis se dissipe.",
                 "Se déplacer vers un lieu de résonance.")
        ELIF version >= 8 AND op_end < 0.6 + 0.05 * (version - 8) THEN
          erreur("StackOverflow", "v" + i2s(version) + " dépasse la capacité de l'opérateur (endurance "
                 + f3(op_end) + ").",
                 "Perte de conscience, vision des Neuf Mondes, risque de folie.",
                 "Un autre RuneSmith doit « couper le lien ».")
        ELSE
          REAL d0 = delta;
          pl("  Niveau " + (version <= 3 | "surface" | (version <= 7 | "profondeur" | "abyssal"))
             + " (v" + i2s(version) + ") ancré à " + lieu + " (Δ local ≈ " + f3(ld[li]) + ").");
          REAL p_ok = (coeur = 3 | 0.97 | (coeur = 2 | 0.78 | 0.55)) - 0.02 * (version > 4 | version - 4 | 0);
          BOOL backlash := FALSE, abandon := FALSE;
          IF alea > p_ok THEN
            IF coeur = 1 THEN
              backlash := TRUE;
              pl("  Cœur Noir : la commande est instable, RETOUR DE FLAMME.");
              op_end := clamp(op_end - 0.10 - 0.01 * version)
            ELSE
              abandon := TRUE;
              pl("  Cœur hésitant : rien ne se passe. Vous avez douté pendant le silence.");
              op_end := clamp(op_end - 0.02)
            FI
          FI;
          IF NOT abandon THEN
            IF alea < clamp(0.03 * version * (1.2 - d0)) THEN
              pl("  ✗ TrameIntercept : Prométhée détecte la brèche.");
              pl("  COMMANDE INVERSÉE (« oubli » devient « souvenir obsessif »).");
              perturber(0.10, -0.06, -0.04); op_end := clamp(op_end - 0.12)
            ELSE
              REAL fc = (coeur = 3 | 1.0 | (coeur = 2 | 0.6 | -0.7));
              REAL amp0 = rp[ri] * (0.04 + 0.02 * version) * fc;
              REAL amp = (backlash | -ABS amp0 | amp0);
              perturber(-amp * 0.8, amp * 0.5, amp * 0.2);
              IF coeur = 1 THEN
                op_end := clamp(op_end - 0.015 * version);
                pl("  Dette ontique contractée : le système la réclamera.")
              FI;
              IF (ri = 10 OR ri = 11) AND version >= 5 AND alea < 0.35 THEN
                pl("  Contamination narrative : un souvenir qui n'est pas le vôtre.")
              FI;
              pl("  " + rn[ri] + " (" + rf[ri] + ") : " + rd[ri]);
              pl("  Δ : " + f3(d0) + " → " + f3(delta) + "   → COMPILÉ")
            FI
          FI
        FI
      FI;
      pl("  [endurance de l'opérateur : " + f3(op_end) + "]")
    END;

  # ----------------------------------------------------------------- #
  # 10. CODEX-STEIN-FRESKO (Fraktal-Gefrickel, vgl. Lisp-Quelle)       #
  # ----------------------------------------------------------------- #
  PROC fresque = (INT prof, phase, indent) VOID:
    BEGIN
      STRING pre = indent * "  ";
      IF prof <= 0 THEN
        pl(pre + "• essence-pure(" + ph_nom[phase] + ")")
      ELSE
        pl(pre + "◆ " + ph_nom[phase] + " — " + ph_ess[phase] + "  (niveau " + i2s(prof) + ")");
        TO 3 DO
          INT sous = 1 + ENTIER (alea * 8.0);
          pl(pre + "  ├─ " + ph_nom[sous] + "-de:");
          fresque(prof - 1, sous, indent + 2)
        OD;
        pl(pre + "  └─ transformation-vers → " + ph_nom[(phase MOD 8) + 1])
      FI
    END;

  # ----------------------------------------------------------------- #
  # 11. HAUPTPROGRAMM: Argumente einlesen (Kommandozeilen-Fummelei)    #
  # ----------------------------------------------------------------- #
  # argv(4..): BEFEHL [SEED|N] [Herz] [Modus] -- die ersten drei Slots sind a68g, Datei und --, also nur Overhead #
  STRING commande := "run";
  INT nombre := 0;
  IF argc >= 4 THEN commande := argv(4) FI;
  IF argc >= 5 THEN nombre := toint(argv(5)) FI;
  IF argc >= 6 THEN
    coeur := (argv(6) = "noir" | 1 | (argv(6) = "blanc" | 3 | 2))
  FI;
  IF argc >= 7 THEN original := (argv(7) = "original") FI;
  IF argc >= 8 THEN cibles_etendues := (argv(8) = "cibles") FI;
  IF commande = "run" THEN
    INT graine = (nombre > 0 | nombre | ENTIER (seconds * 1000.0));
    ignorer_int(run(graine))
  ELIF commande = "mc" THEN
    monte_carlo((nombre > 0 | nombre | 1000), 1)
  ELIF commande = "demo" THEN
    first random((nombre > 0 | nombre | 7)); init; demo
  ELIF commande = "fracturo" THEN
    first random((nombre > 0 | nombre | 7)); init;
    IF argc >= 8 THEN fracturo(argv(8)) ELSE pl("usage : fracturo GRAINE COEUR MODE 'Ω<ᚦ>v4 hague — x •••'") FI
  ELIF commande = "codex" THEN
    first random(7);
    fresque((nombre > 0 AND nombre <= 3 | nombre | 2), 1 + ENTIER (alea * 8.0), 0)
  ELSE
    pl("Commandes : run | mc | demo | fracturo | codex")
  FI
